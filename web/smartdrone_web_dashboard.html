T-SQL Script (Bài Tập Môn Hệ QTCSDL)

Chứa định nghĩa bảng Orders (có ItemName, Category) và Stored Procedures

-- ====================================================================
-- DU AN: SMARTDRONE MANAGEMENT SYSTEM (v3.1)
-- MON HOC: HE QUAN TRI CO SO DU LIEU (SQL SERVER T-SQL)
-- ====================================================================

-- 1. TAO BANG DU LIEU DON HANG (ORDERS) DONG BO THEO YEU CAU MOI
IF OBJECT_ID('Orders', 'U') IS NOT NULL DROP TABLE Orders;
GO

CREATE TABLE Orders (
    OrderID INT IDENTITY(1,1) PRIMARY KEY,
    OrderCode VARCHAR(30) UNIQUE NOT NULL,      -- Ma don hang (VD: ORD_2026_010)
    CustomerName NVARCHAR(100) NOT NULL,        -- Ten khach hang
    ItemName NVARCHAR(150) NOT NULL,            -- Ten hang hoa chi tiet
    Category NVARCHAR(50) NOT NULL,             -- Phan loai: Y te, Linh kien, Tai lieu...
    WeightKg DECIMAL(5,2) NOT NULL,             -- Trong luong (kg)
    PickupStation NVARCHAR(100) NOT NULL,       -- Tram lay hang
    DeliveryStation NVARCHAR(100) NOT NULL,     -- Tram giao hang
    Status VARCHAR(20) DEFAULT 'PENDING',       -- PENDING, APPROVED, REJECTED
    AssignedDroneID INT NULL,                   -- ID Drone duoc gan
    CreatedAt DATETIME DEFAULT GETDATE()
);
GO

-- 2. STORED PROCEDURE: THEM DON HANG MOI (GHI DON HANG)
CREATE PROC sp_ThemDonHang
    @OrderCode VARCHAR(30),
    @CustomerName NVARCHAR(100),
    @ItemName NVARCHAR(150),
    @Category NVARCHAR(50),
    @WeightKg DECIMAL(5,2),
    @PickupStation NVARCHAR(100),
    @DeliveryStation NVARCHAR(100)
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Kiem tra trung ma don hang
    IF EXISTS (SELECT 1 FROM Orders WHERE OrderCode = @OrderCode)
    BEGIN
        RAISERROR(N'Error: Ma don hang da ton tai trong he thong!', 16, 1);
        RETURN;
    END

    -- Ghi don hang moi voi trang thai mac dinh PENDING
    INSERT INTO Orders (OrderCode, CustomerName, ItemName, Category, WeightKg, PickupStation, DeliveryStation, Status)
    VALUES (@OrderCode, @CustomerName, @ItemName, @Category, @WeightKg, @PickupStation, @DeliveryStation, 'PENDING');
    
    PRINT N'Ghi don hang moi thanh cong: ' + @OrderCode;
END;
GO

-- 3. STORED PROCEDURE: DUYET VA PHAN CONG DRONE TU DONG
CREATE PROC sp_DuyetVaPhanCongDonHang
    @OrderID INT,
    @DroneID INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Kiem tra don hang co ton tai va o trang thai PENDING hay khong
    IF NOT EXISTS (SELECT 1 FROM Orders WHERE OrderID = @OrderID AND Status = 'PENDING')
    BEGIN
        RAISERROR(N'Don hang khong hop le hoac da duoc duyyet truoc do!', 16, 1);
        RETURN;
    END

    -- Gia lap tim Drone IDLE va Battery > 20%
    -- Cap nhat trang thai don hang thanh APPROVED va gán Drone
    SET @DroneID = 101; -- Gia lap ID Drone duoc chon

    UPDATE Orders
    SET Status = 'APPROVED',
        AssignedDroneID = @DroneID
    WHERE OrderID = @OrderID;

    PRINT N'Da duyet va gán Drone thanh cong cho don hang ID: ' + CAST(@OrderID AS VARCHAR(10));
END;
GO
